import { Card, CardGrid, Aside } from '@astrojs/starlight/components';

Annota provides a comprehensive set of React hooks for building annotation interfaces.

<Aside type="tip">
  All hooks must be used within an `AnnotaProvider` component.
</Aside>

## Core Hooks

<CardGrid>
  <Card title="useAnnotator" icon="setting">

    [View →](/api/hooks/use-annotator)

    [View →](/api/hooks/use-annotations)
  </Card>

    Get a specific annotation by ID.
    
    [View →](/api/hooks/use-annotation)

## Quick Reference

```tsx
import {
  // Core
  useAnnotator,
  useAnnotations,
  useAnnotation,
  useSelection,
  
  // Tools
  useTool,
  usePushToolCursor,
  
  // Layers
  useLayers,
  useLayer,
  useLayerManager,
  
  // History
  useHistory,
  useCanUndo,
  useCanRedo,
  
  // Interaction
  useHover,
  useEditing,
  useViewer,
  
  // UI
  useContextMenu,
  useContextMenuBinding,
  usePopup,
} from "annota/react";
```
